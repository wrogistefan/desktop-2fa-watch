cmake_minimum_required(VERSION 3.22)
project(d2fa_transfer_poc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static BoringSSL" FORCE)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(D2FA_BORINGSSL_REVISION 594a44291086dcc7cef06cabec4c0b47afd674c0)
set(D2FA_BORINGSSL_SOURCE "" CACHE PATH "Existing checkout of the exact pinned revision")
option(D2FA_TEST_API "Test-only full-key comparison; never ship this build" OFF)
option(D2FA_SANITIZE "Instrument wrapper and BoringSSL with ASan/UBSan" OFF)
option(D2FA_JNI "Build the thin JNI adapter" OFF)
if(D2FA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT D2FA_BORINGSSL_SOURCE)
  message(FATAL_ERROR "Provide D2FA_BORINGSSL_SOURCE at the exact approved PoC revision")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${D2FA_BORINGSSL_SOURCE} rev-parse HEAD
  OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_revision STREQUAL D2FA_BORINGSSL_REVISION)
  message(FATAL_ERROR "BoringSSL checkout differs from the approved PoC pin")
endif()
add_subdirectory(${D2FA_BORINGSSL_SOURCE} boringssl EXCLUDE_FROM_ALL)
add_library(d2fa_transfer SHARED core.cc)
target_include_directories(d2fa_transfer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(d2fa_transfer PRIVATE ssl crypto)
target_compile_definitions(d2fa_transfer PRIVATE D2FA_TRANSFER_BUILD=1)
if(MSVC)
  target_compile_options(d2fa_transfer PRIVATE /W4 /WX)
else()
  target_compile_options(d2fa_transfer PRIVATE -Wall -Wextra -Werror)
endif()
if(APPLE)
  target_link_options(d2fa_transfer PRIVATE
    -Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports-macos.txt)
elseif(UNIX)
  target_link_options(d2fa_transfer PRIVATE -Wl,--exclude-libs,ALL
    -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map)
endif()
if(D2FA_TEST_API)
  target_compile_definitions(d2fa_transfer PRIVATE D2FA_TEST_API=1)
endif()
if(D2FA_JNI)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(d2fa_transfer PRIVATE ${JNI_INCLUDE_DIRS})
  endif()
  target_sources(d2fa_transfer PRIVATE jni.cc)
endif()
enable_testing()
if(D2FA_TEST_API)
  add_executable(d2fa_core_test core_test.cc)
  target_compile_definitions(d2fa_core_test PRIVATE D2FA_TEST_API=1)
  target_link_libraries(d2fa_core_test PRIVATE d2fa_transfer crypto)
  add_test(NAME d2fa_core COMMAND d2fa_core_test)
endif()
